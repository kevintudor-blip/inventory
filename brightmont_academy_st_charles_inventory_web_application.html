Paste this code inside Google Sheets (Extensions > Apps Script) to enable 2-way live sync with spreadsheet 1AVOthk_4lPCPi1zroepEjb19DozSYNyfwP2eifHjNr0. 

// Brightmont Academy St. Charles - 2-Way Sync Bridge Script
const SPREADSHEET_ID = '1AVOthk_4lPCPi1zroepEjb19DozSYNyfwP2eifHjNr0';
const SHEET_GID = '1289365460';

function doGet(e) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheets()[0];
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const rows = data.slice(1).map(row => {
    let item = {};
    headers.forEach((h, i) => { item[h] = row[i]; });
    return item;
  });
  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheets()[0];
    sheet.clearContents();
    
    // 8 Standard Headers
    const headers = [
      "Primary Subject Area",
      "Ordering Name and/or link, if it applies",
      "Description",
      "Quantity",
      "Rooms/Space Storing Them",
      "Contact people",
      "Consumable? Answer \"Yes\" or, if no, leave blank. If \"Yes,\" inform Meredith so that she can replace it.",
      "Other notes"
    ];
    sheet.appendRow(headers);
    
    payload.forEach(item => {
      sheet.appendRow([
        item["Primary Subject Area"] || "",
        item["Ordering Name and/or link, if it applies"] || "",
        item["Description"] || "",
        item["Quantity"] || 0,
        item["Rooms/Space Storing Them"] || "",
        item["Contact people"] || "",
        item["Consumable? Answer \"Yes\" or, if no, leave blank. If \"Yes,\" inform Meredith so that she can replace it."] || "",
        item["Other notes"] || ""
      ]);
    });
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
